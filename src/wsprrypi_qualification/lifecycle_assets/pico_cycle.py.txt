"""Linux/macOS bench adapter using only the production shared USB installer.

This adapter performs no transmission, firmware reset outside maintenance, direct
picotool operation or product-receipt manipulation. Inputs must already be verified.
"""

def ensure(value, message='Lifecycle assertion failed'):
    if not value:
        raise RuntimeError(message)
import argparse
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import subprocess
import sys
parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('operation', choices=('inspect', 'seed', 'verify', 'verify-blank', 'erase', 'install', 'reinstall', 'upgrade', 'recover'))
parser.add_argument('--installer', type=Path, required=True)
parser.add_argument('--installer-sha256', required=True)
parser.add_argument('--schema', type=Path, required=True)
parser.add_argument('--schema-sha256', required=True)
parser.add_argument('--device', required=True)
parser.add_argument('--parent', required=True)
parser.add_argument('--state-dir', type=Path, required=True)
parser.add_argument('--bundle', type=Path, required=True)
parser.add_argument('--baseline-bundle', type=Path)
parser.add_argument('--picotool', type=Path, required=True)
parser.add_argument('--output', type=Path, required=True)
parser.add_argument('--compare', type=Path)
parser.add_argument('--transaction-from', type=Path)
parser.add_argument('--fresh-defaults', action='store_true')
parser.add_argument('--hostname', default='lifecycle-test.local')
args = parser.parse_args()
os.umask(63)
for path in (args.output,args.output.with_suffix('.command.json'),args.output.with_suffix('.blank.bin')):
    ensure(not path.exists() and not path.is_symlink(), 'Fresh Pico evidence destination required before device access')
ensure(not any(path.is_symlink() for path in args.output.parents), 'Symlink Pico evidence parent refused')
args.output.parent.mkdir(mode=448,parents=True,exist_ok=True)
metadata=args.output.parent.stat()
ensure(metadata.st_uid==os.getuid() and metadata.st_mode&0o077==0, 'Private Pico evidence directory required')

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def save(path, value):
    with path.open('x', encoding='utf-8') as stream:
        json.dump(value, stream, indent=2)
        stream.flush()
        os.fsync(stream.fileno())
    if os.name=='posix':
        directory=os.open(path.parent,os.O_RDONLY|os.O_DIRECTORY)
        try:os.fsync(directory)
        finally:os.close(directory)
ensure(sha(args.installer) == args.installer_sha256, 'Installer changed')
ensure(sha(args.schema) == args.schema_sha256, 'Schema changed')
sys.dont_write_bytecode = True
spec = importlib.util.spec_from_file_location('shared_installer', args.installer)
core = importlib.util.module_from_spec(spec)
spec.loader.exec_module(core)
core.SCHEMA_BYTES = args.schema.read_bytes()
files = core.read_bundle(args.bundle)
local = 'local-build.json' in files
manifest, _ = core.verify_local(files) if local else core.verify_bundle(files, 'devel')
if args.baseline_bundle:
    old = core.read_bundle(args.baseline_bundle)
    core.verify_local(old) if 'local-build.json' in old else core.verify_bundle(old, 'devel')
core.prune_transactions(args.state_dir, retain_history=True, admit=False)
board = core.selected(args.device, parent=args.parent)
result = {'serial': args.device, 'parent': args.parent, 'operation': args.operation, 'candidate_source': manifest['source']['revision'], 'rf_enabled': False, 'bundle_sha256': {name: hashlib.sha256(data).hexdigest() for name, data in files.items()}}
if args.operation == 'verify-blank':
    with core.device_lock(args.device):
        ensure(board['mode'] == 'bootsel', 'Lifecycle assertion failed')
        tool = core.Picotool(str(args.picotool))
        info = tool.run(args.device, 'info', '-a')
        import re
        ensure(re.search('^\\s+type:\\s+RP2350$', info, re.M), 'Lifecycle assertion failed')
        ensure(re.search('^\\s+chipid:\\s+0x' + args.device.lower() + '$', info, re.M), 'Lifecycle assertion failed')
        blank = tool.read_flash(args.device, args.output.with_suffix('.blank.bin'))
        ensure(len(blank) == core.FLASH_SIZE and blank == b'\xff' * core.FLASH_SIZE, 'Lifecycle assertion failed')
        result.update(verified=True, blank_sha256=hashlib.sha256(blank).hexdigest())
elif args.operation in ('inspect', 'verify', 'seed'):
    with core.device_lock(args.device):
        snapshot = core.public_snapshot(board)
        core.admit_snapshot(snapshot)
        ensure(snapshot['config'].get('config') is None or snapshot['config']['config']['enabled'] is False, 'Lifecycle assertion failed')
        if args.operation == 'seed':
            with core.Management(board) as management:
                path = '/api/control/v1/hostname'
                current = management.get(path)
                body = {'contract': 'Wsprry-Control/1', 'schema': 'hostname', 'target': snapshot['target'], 'hostname': args.hostname}
                reply = management.exchange(dict(operation='request', method='PUT', path=path, body=json.dumps(body, separators=(',', ':')), if_match=management.etags[path]))
                ensure(reply['delivery_required'] is True, 'Lifecycle assertion failed')
                ack = management.exchange(dict(operation='ack'), repeat=True)
                ensure(reply['status'] == 200 and ack['status'] == 200, 'Settings refused')
            snapshot = core.public_snapshot(board)
            ensure(snapshot['hostname']['saved'] == args.hostname, 'Lifecycle assertion failed')
        if args.operation == 'verify':
            ensure(snapshot['firmware'] == manifest['firmware_version'], 'Lifecycle assertion failed')
            ensure(snapshot['source_revision'] == manifest['source'].get('identity', manifest['source']['revision'][:12]), 'Lifecycle assertion failed')
        if args.fresh_defaults:
            ensure(snapshot['config'].get('config') is None, 'Old standalone configuration remains')
            ensure(not snapshot['progress']['fields'], 'Old progress records remain')
        semantics = core.saved_semantics(snapshot)
        if args.compare:
            previous = json.loads(args.compare.read_text())
            ensure(semantics == previous['semantics'], 'Saved settings changed')
        result.update(target=snapshot['target'], firmware=snapshot['firmware'], source=snapshot['source_revision'], semantics=semantics, safe=True, verified=True)
else:
    operation = 'upgrade' if args.operation in ('upgrade', 'reinstall') else args.operation
    command = [sys.executable, str(args.installer), operation, '--device', args.device, '--board', 'pico2_w', '--yes', '--json-only', '--picotool', str(args.picotool), '--state-dir', str(args.state_dir)]
    if args.operation == 'recover':
        ensure(args.transaction_from is not None, 'Lifecycle assertion failed')
        receipt = json.loads(args.transaction_from.read_text())
        ensure(receipt['serial'] == args.device and receipt['parent'] == args.parent, 'Lifecycle assertion failed')
        transaction=receipt.get('transaction')
        if transaction is None:
            transaction=json.loads(receipt['stdout']).get('transaction')
        ensure(isinstance(transaction,str), 'Missing production recovery transaction')
        command += ['--transaction', transaction]
    elif args.operation == 'erase':
        command += ['--software-bootsel']
    else:
        command += ['--local-bundle' if local else '--bundle', str(args.bundle)]
        if operation == 'upgrade' and args.baseline_bundle:
            command += ['--installed-bundle', str(args.baseline_bundle)]
    execution = subprocess.run(command, capture_output=True, text=True, timeout=600)
    result.update(command=command, returncode=execution.returncode, stdout=execution.stdout, stderr=execution.stderr)
    save(args.output.with_suffix('.command.json'), result)
    ensure(execution.returncode == 0, 'Production installer failed; inspect retained command/transaction')
    observed = json.loads(execution.stdout)
    ensure(observed['stage'] == 'complete' and observed.get('recovery_required') is False, 'Lifecycle assertion failed')
    if args.operation == 'erase':
        ensure(observed['erase_verified'] is True and observed['erased_bytes'] == 4 * 1024 * 1024, 'Lifecycle assertion failed')
    result.update(transaction=observed['transaction'], verified=True)
save(args.output, result)
print(json.dumps({'verified': True, 'operation': args.operation, 'serial': args.device, 'evidence_sha256': sha(args.output), 'observation': result}))
